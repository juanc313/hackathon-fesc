    raw = os.environ.get("WORKER_INTERVAL_SECONDS", "")

        return int(raw)

== Reproducción del fallo y verificación de la corrección ==
# En el servidor Linux del evento el intérprete se llama 'python3'; en este equipo
# de Windows 'python3' es el alias inservible de la Microsoft Store (no ejecuta
# nada), así que los comandos de esta sección se corrieron con 'py -3' (Python
# 3.14.8). No hace falta ningún archivo auxiliar: el sustituto del driver va en
# línea, porque lo único que worker.py necesita de psycopg2 para llegar a
# load_interval() es el nombre del módulo y el de OperationalError.

$ WORKER_INTERVAL_SECONDS=30s py -3 - <<'PY'

worker.load_interval()
